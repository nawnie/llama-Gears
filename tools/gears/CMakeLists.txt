if (LLAMA_GEARS_INSPECT)
    add_library(llama-gears-policy STATIC src/policy.cpp)
    target_include_directories(llama-gears-policy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_features(llama-gears-policy PUBLIC cxx_std_17)

    add_executable(llama-gears-inspect src/inspect.cpp)
    target_link_libraries(llama-gears-inspect PRIVATE llama-gears-policy ggml)
    target_compile_features(llama-gears-inspect PRIVATE cxx_std_17)

    if (LLAMA_BUILD_TESTS)
        add_executable(llama-gears-policy-test tests/policy_test.cpp)
        target_link_libraries(llama-gears-policy-test PRIVATE llama-gears-policy)
        add_test(NAME llama-gears-policy COMMAND llama-gears-policy-test)
    endif()
endif()

if (LLAMA_GEARS_RUNTIME)
    if (NOT BUILD_SHARED_LIBS)
        message(FATAL_ERROR "Gears runtime requires BUILD_SHARED_LIBS=ON to use the same GGML core")
    endif()
    add_library(ggml-gears MODULE runtime/ggml-gears.cpp)
    target_include_directories(ggml-gears PRIVATE ${PROJECT_SOURCE_DIR}/ggml/src)
    target_compile_definitions(ggml-gears PRIVATE GGML_BACKEND_DL GGML_BACKEND_SHARED GGML_BACKEND_BUILD)
    target_link_libraries(ggml-gears PRIVATE ggml Threads::Threads)
    target_compile_features(ggml-gears PRIVATE cxx_std_17)

    if (LLAMA_BUILD_TESTS)
        target_compile_definitions(ggml-gears PRIVATE LLAMA_GEARS_RUNTIME_TESTING)
        add_executable(llama-gears-runtime-cache-test tests/runtime_cache_test.cpp)
        target_include_directories(llama-gears-runtime-cache-test PRIVATE runtime)
        target_compile_features(llama-gears-runtime-cache-test PRIVATE cxx_std_17)
        add_test(NAME llama-gears-runtime-cache COMMAND llama-gears-runtime-cache-test)

        add_executable(llama-gears-runtime-test tests/runtime_backend_test.cpp)
        target_include_directories(llama-gears-runtime-test PRIVATE runtime)
        target_link_libraries(llama-gears-runtime-test PRIVATE ggml)
        target_compile_features(llama-gears-runtime-test PRIVATE cxx_std_17)
        add_dependencies(llama-gears-runtime-test ggml-gears)
        add_test(NAME llama-gears-runtime COMMAND llama-gears-runtime-test $<TARGET_FILE:ggml-gears>)
    endif()
endif()
